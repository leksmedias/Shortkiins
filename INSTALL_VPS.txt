OPTION 2: Manual Installation (Copy-Paste All Commands)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

# Install dependencies
sudo apt update && sudo apt install -y python3 python3-pip python3-venv ffmpeg git

# Clone repository from feature branch
git clone -b claude/video-generator-pipeline-fal1T https://github.com/leksmedias/Shortkiins.git
cd Shortkiins

# Set up Python environment
python3 -m venv venv
source venv/bin/activate
pip install --upgrade pip
pip install -r requirements.txt

# Configure API keys
cp .env.example .env
echo ""
echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
echo "⚠️  IMPORTANT: Configure your API keys now!"
echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
echo ""
echo "Edit .env file and add these REQUIRED keys:"
echo "1. INWORLD_API_KEY (base64 credential from https://studio.inworld.ai/)"
echo "2. GROQ_API_KEY (free from https://console.groq.com/)"
echo "3. WAVESPEED_API_KEY (from https://wavespeed.ai/)"
echo ""
nano .env

# Create directories
mkdir -p output temp static

# Test installation
python main.py --help

# Start web interface
echo ""
echo "✅ Installation complete! Starting web interface..."
echo "Access at: http://YOUR_SERVER_IP:8000"
python api.py
